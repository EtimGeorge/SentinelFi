// frontend/components/context/AuthContext.tsx
import React, { createContext, useContext, useState, useEffect, useCallback, useMemo } from 'react';
import { useRouter } from 'next/router';
import axios from 'axios';

// ==================== TYPES ====================
export enum Role {
  SuperAdmin = 'SuperAdmin',
  TenantAdmin = 'TenantAdmin',
  TenantUser = 'TenantUser',
}

export interface User {
  id: string;
  email: string;
  role: Role;
  tenantId?: string;
  firstName?: string;
  lastName?: string;
  permissions?: string[];
}

export interface LoginCredentials {
  email: string;
  password: string;
  tenantId?: string;
}

export interface AuthState {
  user: User | null;
  token: string | null;
  isAuthenticated: boolean;
  isLoading: boolean;
  error: string | null;
  loginAttempts: number;
  lastLoginAttempt: Date | null;
}

export interface AuthContextType extends AuthState {
  login: (credentials: LoginCredentials) => Promise<{ success: boolean; redirectUrl?: string }>;
  logout: () => Promise<void>;
  refreshUser: () => Promise<void>;
  clearError: () => void;
  hasPermission: (permission: string) => boolean;
  hasRole: (role: Role | Role[]) => boolean;
}

// ==================== CONSTANTS ====================
const AUTH_TOKEN_KEY = 'sentinelfi_auth_token';
const AUTH_USER_KEY = 'sentinelfi_user';
const MAX_LOGIN_ATTEMPTS = 5;
const LOGIN_ATTEMPT_WINDOW_MS = 15 * 60 * 1000; // 15 minutes

const API_BASE_URL = process.env.NEXT_PUBLIC_API_URL || 'http://localhost:3000';

// ==================== ROUTE MAPPING ====================
export const ROLE_ROUTES: Record<Role, string> = {
  [Role.SuperAdmin]: '/super/index',
  [Role.TenantAdmin]: '/dashboard/home',
  [Role.TenantUser]: '/dashboard/home',
};

export const PUBLIC_ROUTES = ['/login', '/register', '/forgot-password', '/reset-password'];
export const AUTH_ROUTES = ['/login', '/register'];

// ==================== CONTEXT ====================
const AuthContext = createContext<AuthContextType | undefined>(undefined);

// ==================== HELPER FUNCTIONS ====================
class AuthStorage {
  static setToken(token: string): void {
    if (typeof window !== 'undefined') {
      localStorage.setItem(AUTH_TOKEN_KEY, token);
      axios.defaults.headers.common['Authorization'] = `Bearer ${token}`;
    }
  }

  static getToken(): string | null {
    if (typeof window !== 'undefined') {
      return localStorage.getItem(AUTH_TOKEN_KEY);
    }
    return null;
  }

  static setUser(user: User): void {
    if (typeof window !== 'undefined') {
      localStorage.setItem(AUTH_USER_KEY, JSON.stringify(user));
    }
  }

  static getUser(): User | null {
    if (typeof window !== 'undefined') {
      const userStr = localStorage.getItem(AUTH_USER_KEY);
      if (userStr) {
        try {
          return JSON.parse(userStr) as User;
        } catch (e) {
          console.error('[AuthStorage] Failed to parse user:', e);
          return null;
        }
      }
    }
    return null;
  }

  static clear(): void {
    if (typeof window !== 'undefined') {
      localStorage.removeItem(AUTH_TOKEN_KEY);
      localStorage.removeItem(AUTH_USER_KEY);
      delete axios.defaults.headers.common['Authorization'];
    }
  }
}

class AuthLogger {
  private static prefix = '[AUTH]';

  static info(message: string, data?: any): void {
    console.log(`${this.prefix} ${message}`, data || '');
  }

  static error(message: string, error?: any): void {
    console.error(`${this.prefix} ERROR: ${message}`, error || '');
  }

  static warn(message: string, data?: any): void {
    console.warn(`${this.prefix} WARNING: ${message}`, data || '');
  }

  static loginAttempt(email: string, role?: Role): void {
    this.info(`Login attempt - Email: ${email}, Role: ${role || 'N/A'}`);
  }

  static loginSuccess(user: User): void {
    this.info(`Login SUCCESS - User: ${user.email}, Role: ${user.role}, ID: ${user.id}`);
  }

  static loginFailure(email: string, reason: string): void {
    this.error(`Login FAILED - Email: ${email}, Reason: ${reason}`);
  }

  static redirecting(from: string, to: string, role: Role): void {
    this.info(`Redirecting - From: ${from}, To: ${to}, Role: ${role}`);
  }
}

// ==================== PROVIDER COMPONENT ====================
export const AuthProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const router = useRouter();

  const [authState, setAuthState] = useState<AuthState>({
    user: null,
    token: null,
    isAuthenticated: false,
    isLoading: true,
    error: null,
    loginAttempts: 0,
    lastLoginAttempt: null,
  });

  // ==================== INITIALIZATION ====================
  useEffect(() => {
    const initAuth = async () => {
      AuthLogger.info('Initializing authentication...');
      
      const token = AuthStorage.getToken();
      const user = AuthStorage.getUser();

      if (token && user) {
        AuthLogger.info('Found existing session', { email: user.email, role: user.role });
        
        // Set axios default header
        axios.defaults.headers.common['Authorization'] = `Bearer ${token}`;

        // Validate token with backend
        try {
          const response = await axios.get(`${API_BASE_URL}/api/v1/auth/validate`);
          if (response.data.valid) {
            setAuthState({
              user,
              token,
              isAuthenticated: true,
              isLoading: false,
              error: null,
              loginAttempts: 0,
              lastLoginAttempt: null,
            });
            AuthLogger.info('Session validated successfully');
          } else {
            throw new Error('Token validation failed');
          }
        } catch (error) {
          AuthLogger.warn('Token validation failed, clearing session');
          AuthStorage.clear();
          setAuthState(prev => ({ ...prev, isLoading: false }));
        }
      } else {
        AuthLogger.info('No existing session found');
        setAuthState(prev => ({ ...prev, isLoading: false }));
      }
    };

    initAuth();
  }, []);

  // ==================== LOGIN ====================
  const login = useCallback(async (credentials: LoginCredentials): Promise<{ success: boolean; redirectUrl?: string }> => {
    AuthLogger.loginAttempt(credentials.email);

    // Rate limiting check
    if (authState.loginAttempts >= MAX_LOGIN_ATTEMPTS) {
      const timeSinceLastAttempt = authState.lastLoginAttempt 
        ? Date.now() - authState.lastLoginAttempt.getTime()
        : Infinity;

      if (timeSinceLastAttempt < LOGIN_ATTEMPT_WINDOW_MS) {
        const waitMinutes = Math.ceil((LOGIN_ATTEMPT_WINDOW_MS - timeSinceLastAttempt) / 60000);
        const errorMsg = `Too many login attempts. Please try again in ${waitMinutes} minutes.`;
        AuthLogger.warn(errorMsg);
        setAuthState(prev => ({ ...prev, error: errorMsg }));
        return { success: false };
      } else {
        // Reset counter after window expires
        setAuthState(prev => ({ ...prev, loginAttempts: 0 }));
      }
    }

    setAuthState(prev => ({ ...prev, isLoading: true, error: null }));

    try {
      // Construct login endpoint based on role
      const endpoint = credentials.tenantId 
        ? `${API_BASE_URL}/api/v1/auth/login/tenant`
        : `${API_BASE_URL}/api/v1/auth/login/super`;

      const payload = credentials.tenantId 
        ? { email: credentials.email, password: credentials.password, tenantId: credentials.tenantId }
        : { email: credentials.email, password: credentials.password };

      AuthLogger.info(`Sending login request to: ${endpoint}`);

      const response = await axios.post(endpoint, payload);

      const { token, user } = response.data;

      if (!token || !user) {
        throw new Error('Invalid response from server: missing token or user');
      }

      // Validate user object
      if (!user.id || !user.email || !user.role) {
        throw new Error('Invalid user object received from server');
      }

      // Store credentials
      AuthStorage.setToken(token);
      AuthStorage.setUser(user);

      // Update state
      setAuthState({
        user,
        token,
        isAuthenticated: true,
        isLoading: false,
        error: null,
        loginAttempts: 0,
        lastLoginAttempt: null,
      });

      AuthLogger.loginSuccess(user);

      // Determine redirect URL based on role
      const redirectUrl = ROLE_ROUTES[user.role as Role] || '/dashboard/home';
      AuthLogger.info(`Login successful. Redirect URL: ${redirectUrl}`);

      return { success: true, redirectUrl };

    } catch (error: any) {
      const errorMessage = error.response?.data?.message || error.message || 'Login failed';
      
      AuthLogger.loginFailure(credentials.email, errorMessage);

      setAuthState(prev => ({
        ...prev,
        isLoading: false,
        error: errorMessage,
        loginAttempts: prev.loginAttempts + 1,
        lastLoginAttempt: new Date(),
      }));

      return { success: false };
    }
  }, [authState.loginAttempts, authState.lastLoginAttempt]);

  // ==================== LOGOUT ====================
  const logout = useCallback(async () => {
    AuthLogger.info('Logging out...');

    try {
      // Notify backend (optional, for token revocation)
      await axios.post(`${API_BASE_URL}/api/v1/auth/logout`).catch(() => {
        // Ignore errors on logout
      });
    } finally {
      AuthStorage.clear();
      setAuthState({
        user: null,
        token: null,
        isAuthenticated: false,
        isLoading: false,
        error: null,
        loginAttempts: 0,
        lastLoginAttempt: null,
      });

      AuthLogger.info('Logout complete, redirecting to login...');
      await router.push('/login');
    }
  }, [router]);

  // ==================== REFRESH USER ====================
  const refreshUser = useCallback(async () => {
    if (!authState.token) {
      AuthLogger.warn('Cannot refresh user: no token');
      return;
    }

    try {
      const response = await axios.get(`${API_BASE_URL}/api/v1/auth/me`);
      const user = response.data;

      AuthStorage.setUser(user);
      setAuthState(prev => ({ ...prev, user }));
      
      AuthLogger.info('User data refreshed');
    } catch (error) {
      AuthLogger.error('Failed to refresh user', error);
      // Don't logout on refresh failure, token might still be valid
    }
  }, [authState.token]);

  // ==================== UTILITY FUNCTIONS ====================
  const clearError = useCallback(() => {
    setAuthState(prev => ({ ...prev, error: null }));
  }, []);

  const hasPermission = useCallback((permission: string): boolean => {
    if (!authState.user) return false;
    if (authState.user.role === Role.SuperAdmin) return true; // SuperAdmin has all permissions
    return authState.user.permissions?.includes(permission) || false;
  }, [authState.user]);

  const hasRole = useCallback((role: Role | Role[]): boolean => {
    if (!authState.user) return false;
    const roles = Array.isArray(role) ? role : [role];
    return roles.includes(authState.user.role as Role);
  }, [authState.user]);

  // ==================== CONTEXT VALUE ====================
  const contextValue = useMemo<AuthContextType>(() => ({
    ...authState,
    login,
    logout,
    refreshUser,
    clearError,
    hasPermission,
    hasRole,
  }), [authState, login, logout, refreshUser, clearError, hasPermission, hasRole]);

  return (
    <AuthContext.Provider value={contextValue}>
      {children}
    </AuthContext.Provider>
  );
};

// ==================== HOOK ====================
export const useAuth = (): AuthContextType => {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error('useAuth must be used within an AuthProvider');
  }
  return context;
};

export default AuthContext;
