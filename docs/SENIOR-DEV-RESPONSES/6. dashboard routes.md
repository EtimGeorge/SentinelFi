// backend/src/routes/dashboard.routes.ts
import { Router } from 'express';
import AuthMiddleware from '../middleware/auth.middleware';
import { DashboardController } from '../controllers/dashboard.controller';

const router = Router();
const dashboardController = new DashboardController();

/**
 * CRITICAL: This route was returning 500 errors because SuperAdmin users
 * were accessing it without proper validation.
 * 
 * Solution: Only allow tenant users (TenantAdmin and TenantUser) to access
 * tenant-specific endpoints. SuperAdmin has their own separate endpoints.
 */
router.get(
  '/summary',
  ...AuthMiddleware.tenantUsers, // authenticate + authorize(TenantAdmin, TenantUser) + enforceTenantScope
  dashboardController.getSummary.bind(dashboardController)
);

router.get(
  '/analytics',
  ...AuthMiddleware.tenantUsers,
  dashboardController.getAnalytics.bind(dashboardController)
);

router.get(
  '/recent-activity',
  ...AuthMiddleware.tenantUsers,
  dashboardController.getRecentActivity.bind(dashboardController)
);

export default router;


// backend/src/routes/super-admin.routes.ts
import { Router } from 'express';
import AuthMiddleware from '../middleware/auth.middleware';
import { SuperAdminController } from '../controllers/super-admin.controller';

const router = Router();
const superAdminController = new SuperAdminController();

/**
 * SuperAdmin Routes - Only accessible by SuperAdmin role

 */
router.get(
  '/tenants',
  ...AuthMiddleware.superAdminOnly,
  superAdminController.getAllTenants.bind(superAdminController)
);

  '/tenants/:tenantId',
  ...AuthMiddleware.superAdminOnly,
  superAdminController.getTenantById.bind(superAdminController)
);

router.post(
  '/tenants',
  ...AuthMiddleware.superAdminOnly,
  superAdminController.createTenant.bind(superAdminController)
);

router.put(
  '/tenants/:tenantId',
  ...AuthMiddleware.superAdminOnly,
  superAdminController.updateTenant.bind(superAdminController)
);

router.delete(
  '/tenants/:tenantId',
  ...AuthMiddleware.superAdminOnly,
  superAdminController.deleteTenant.bind(superAdminController)
);

// SuperAdmin dashboard - aggregated data across ALL tenants
router.get(
  '/dashboard/global-summary',
  ...AuthMiddleware.superAdminOnly,
  superAdminController.getGlobalSummary.bind(superAdminController)
);

router.get(
  '/users',
  ...AuthMiddleware.superAdminOnly,
  superAdminController.getAllUsers.bind(superAdminController)
);

export default router;


// backend/src/routes/auth.routes.ts
import { Router } from 'express';
import { AuthController } from '../controllers/auth.controller';
import AuthMiddleware from '../middleware/auth.middleware';

const router = Router();
const authController = new AuthController();

// Public routes (no authentication required)
router.post('/login/super', authController.loginSuperAdmin.bind(authController));
router.post('/login/tenant', authController.loginTenant.bind(authController));
router.post('/register', authController.register.bind(authController));
router.post('/forgot-password', authController.forgotPassword.bind(authController));
router.post('/reset-password', authController.resetPassword.bind(authController));

// Protected routes (authentication required)
router.get(
  '/validate',
  ...AuthMiddleware.authenticated,
  authController.validateToken.bind(authController)
);

router.get(
  '/me',
  ...AuthMiddleware.authenticated,
  authController.getCurrentUser.bind(authController)
);

router.post(
  '/logout',
  ...AuthMiddleware.authenticated,
  authController.logout.bind(authController)
);

router.post(
  '/refresh',
  ...AuthMiddleware.authenticated,
  authController.refreshToken.bind(authController)
);

export default router;


// backend/src/app.ts - Main Express app setup
import express from 'express';
import cors from 'cors';
import authRoutes from './routes/auth.routes';
import dashboardRoutes from './routes/dashboard.routes';
import superAdminRoutes from './routes/super-admin.routes';

const app = express();

// Middleware
app.use(cors());
app.use(express.json());

// Request logging
app.use((req, res, next) => {
  console.log(`[${new Date().toISOString()}] ${req.method} ${req.path}`);
  next();
});

// Routes
app.use('/api/v1/auth', authRoutes);
app.use('/api/v1/dashboard', dashboardRoutes); // Tenant-only endpoints
app.use('/api/v1/super', superAdminRoutes);     // SuperAdmin-only endpoints

// Health check
app.get('/health', (req, res) => {
  res.json({ status: 'ok', timestamp: new Date().toISOString() });
});

// 404 handler
app.use((req, res) => {
  res.status(404).json({ 
    success: false,
    message: 'Endpoint not found',
    path: req.path 
  });
});

// Error handler
app.use((err: any, req: express.Request, res: express.Response, next: express.NextFunction) => {
  console.error('[Error Handler]', err);
  res.status(err.status || 500).json({ 
    success: false,
    message: err.message || 'Internal server error',
    ...(process.env.NODE_ENV === 'development' && { stack: err.stack })
  });
});

export default app;
