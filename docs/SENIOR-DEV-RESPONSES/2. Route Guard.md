// frontend/components/guards/RouteGuard.tsx
import { useEffect, useState } from 'react';
import { useRouter } from 'next/router';
import { useAuth, Role, ROLE_ROUTES, PUBLIC_ROUTES } from '../context/AuthContext';

interface RouteGuardProps {
  children: React.ReactNode;
}

/**
 * RouteGuard - Protects routes based on authentication and role
 * 
 * Rules:
 * 1. Public routes (/login, /register, etc.) - accessible to everyone
 * 2. SuperAdmin routes (/super/*) - only SuperAdmin
 * 3. Tenant routes (/dashboard/*) - TenantAdmin and TenantUser
 * 4. Authenticated routes - redirect to login if not authenticated
 * 5. Wrong role - redirect to correct dashboard
 */
export const RouteGuard: React.FC<RouteGuardProps> = ({ children }) => {
  const router = useRouter();
  const { user, isAuthenticated, isLoading } = useAuth();
  const [authorized, setAuthorized] = useState(false);

  useEffect(() => {
    // Don't run authorization check while auth is loading
    if (isLoading) {
      console.log('[RouteGuard] Auth is loading, waiting...');
      return;
    }

    const checkAuth = () => {
      const path = router.pathname;
      console.log('[RouteGuard] Checking authorization for path:', path);
      console.log('[RouteGuard] User:', user ? { email: user.email, role: user.role } : 'null');
      console.log('[RouteGuard] isAuthenticated:', isAuthenticated);

      // PUBLIC ROUTES - always allow
      if (PUBLIC_ROUTES.includes(path)) {
        console.log('[RouteGuard] Public route, allowing access');
        
        // If user is already logged in and trying to access login/register, redirect to their dashboard
        if (isAuthenticated && user) {
          const userDashboard = ROLE_ROUTES[user.role as Role];
          console.log('[RouteGuard] User already authenticated, redirecting to:', userDashboard);
          router.replace(userDashboard);
          return;
        }
        
        setAuthorized(true);
        return;
      }

      // NOT AUTHENTICATED - redirect to login
      if (!isAuthenticated || !user) {
        console.log('[RouteGuard] Not authenticated, redirecting to login');
        router.replace({
          pathname: '/login',
          query: { returnUrl: router.asPath }
        });
        return;
      }

      // ROLE-BASED ACCESS CONTROL
      const userRole = user.role as Role;
      
      // SuperAdmin trying to access tenant routes
      if (userRole === Role.SuperAdmin && path.startsWith('/dashboard')) {
        console.log('[RouteGuard] SuperAdmin accessing tenant route, redirecting to super dashboard');
        router.replace(ROLE_ROUTES[Role.SuperAdmin]);
        return;
      }

      // Tenant users trying to access SuperAdmin routes
      if ((userRole === Role.TenantAdmin || userRole === Role.TenantUser) && path.startsWith('/super')) {
        console.log('[RouteGuard] Tenant user accessing super route, redirecting to tenant dashboard');
        router.replace(ROLE_ROUTES[userRole]);
        return;
      }

      // All checks passed
      console.log('[RouteGuard] Authorization successful');
      setAuthorized(true);
    };

    checkAuth();

    // Re-run check on route change
    router.events.on('routeChangeStart', checkAuth);
    return () => {
      router.events.off('routeChangeStart', checkAuth);
    };
  }, [router, user, isAuthenticated, isLoading]);

  // Show loading state while checking authorization
  if (isLoading || !authorized) {
    return (
      <div style={{
        display: 'flex',
        justifyContent: 'center',
        alignItems: 'center',
        height: '100vh',
        backgroundColor: '#f5f5f5'
      }}>
        <div style={{ textAlign: 'center' }}>
          <div style={{
            border: '4px solid #f3f3f3',
            borderTop: '4px solid #3498db',
            borderRadius: '50%',
            width: '40px',
            height: '40px',
            animation: 'spin 1s linear infinite',
            margin: '0 auto 16px'
          }} />
          <p style={{ color: '#666', fontSize: '14px' }}>
            {isLoading ? 'Authenticating...' : 'Authorizing...'}
          </p>
        </div>
        <style jsx>{`
          @keyframes spin {
            0% { transform: rotate(0deg); }
            100% { transform: rotate(360deg); }
          }
        `}</style>
      </div>
    );
  }

  return <>{children}</>;
};

export default RouteGuard;
