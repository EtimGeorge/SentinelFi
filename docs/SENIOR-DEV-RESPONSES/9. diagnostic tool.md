        const API_URL = 'http://localhost:3000';
        
        function getStatusHtml(passed, message, details = '') {
            const type = passed ? 'pass' : 'fail';
            const icon = passed 
                ? '<svg class="status-icon" fill="none" stroke="#10b981" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M5 13l4 4L19 7"></path></svg>'
                : '<svg class="status-icon" fill="none" stroke="#ef4444" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M6 18L18 6M6 6l12 12"></path></svg>';
            
            return `
                <div class="status status-${type}">
                    ${icon}
                    <div class="status-text">
                        <div>${message}</div>
                        ${details ? `<div class="detail">${details}</div>` : ''}
                    </div>
                </div>
            `;
        }
        
        function runDiagnostics() {
            const checks = [];
            
            // Check 1: LocalStorage
            const token = localStorage.getItem('sentinelfi_auth_token');
            const userStr = localStorage.getItem('sentinelfi_user');
            
            checks.push(getStatusHtml(
                !!token,
                'Auth Token',
                token ? `Token exists (${token.length} chars)` : 'No token found in localStorage'
            ));
            
            checks.push(getStatusHtml(
                !!userStr,
                'User Data',
                userStr ? 'User data exists in localStorage' : 'No user data found'
            ));
            
            // Check 2: User Object Validation
            if (userStr) {
                try {
                    const user = JSON.parse(userStr);
                    const hasRole = !!user.role;
                    const hasEmail = !!user.email;
                    const hasId = !!user.id;
                    
                    checks.push(getStatusHtml(
                        hasRole && hasEmail && hasId,
                        'User Object Structure',
                        `Role: <span class="highlight">${user.role || 'MISSING'}</span>, Email: ${user.email || 'MISSING'}, ID: ${user.id || 'MISSING'}`
                    ));
                    
                    if (user.role) {
                        const validRoles = ['SuperAdmin', 'TenantAdmin', 'TenantUser'];
                        const roleValid = validRoles.includes(user.role);
                        checks.push(getStatusHtml(
                            roleValid,
                            'Role Validation',
                            roleValid ? `"${user.role}" is valid` : `"${user.role}" is NOT a valid role (must be: ${validRoles.join(', ')})`
                        ));
                    }
                    
                    if (user.role !== 'SuperAdmin') {
                        checks.push(getStatusHtml(
                            !!user.tenantId,
                            'Tenant ID (Required for non-SuperAdmin)',
                            user.tenantId ? `Tenant ID: ${user.tenantId}` : 'MISSING - This will cause 500 errors!'
                        ));
                    }
                } catch (e) {
                    checks.push(getStatusHtml(
                        false,
                        'User Object Parsing',
                        `Error: ${e.message}`
                    ));
                }
            }
            
            // Check 3: JWT Structure
            if (token) {
                try {
                    const parts = token.split('.');
                    if (parts.length === 3) {
                        const payload = JSON.parse(atob(parts[1]));
                        checks.push(getStatusHtml(
                            true,
                            'JWT Structure',
                            `Valid JWT with ${Object.keys(payload).length} claims`
                        ));
                        
                        // Check expiration
                        if (payload.exp) {
                            const expDate = new Date(payload.exp * 1000);
                            const isExpired = expDate < new Date();
                            checks.push(getStatusHtml(
                                !isExpired,
                                'Token Expiration',
                                isExpired ? `EXPIRED on ${expDate.toLocaleString()}` : `Valid until ${expDate.toLocaleString()}`
                            ));
                        }
                    } else {
                        checks.push(getStatusHtml(
                            false,
                            'JWT Structure',
                            `Invalid JWT format (${parts.length} parts instead of 3)`
                        ));
                    }
                } catch (e) {
                    checks.push(getStatusHtml(
                        false,
                        'JWT Decoding',
                        `Error: ${e.message}`
                    ));
                }
            }
            
            // Display results
            document.getElementById('quickChecks').innerHTML = checks.join('');
            
            // Update auth state
            displayAuthState();
        }
        
        function displayAuthState() {
            const token = localStorage.getItem('sentinelfi_auth_token');
            const userStr = localStorage.getItem('sentinelfi_user');
            
            if (!token || !userStr) {
                document.getElementById('authState').innerHTML = `
                    <div class="status status-warn">
                        <div class="status-text">Not authenticated</div>
                    </div>
                `;
                return;
            }
            
            try {
                const user = JSON.parse(userStr);
                const tokenParts = token.split('.');
                const payload = JSON.parse(atob(tokenParts[1]));
                
                document.getElementById('authState').innerHTML = `
                    <div class="code-block">
<strong>Stored User:</strong>
${JSON.stringify(user, null, 2)}

<strong>JWT Payload:</strong>
${JSON.stringify(payload, null, 2)}
                    </div>
                `;
            } catch (e) {
                document.getElementById('authState').innerHTML = `
                    <div class="status status-fail">
                        <div class="status-text">Error parsing auth data: ${e.message}</div>
                    </div>
                `;
            }
        }
        
        async function testBackend() {
            const statusDiv = document.getElementById('backendStatus');
            statusDiv.innerHTML = '<p>Testing backend connection...</p>';
            
            const tests = [];
            
            // Test 1: Health check
            try {
                const healthRes = await fetch(`${API_URL}/health`);
                const healthData = await healthRes.json();
                tests.push(getStatusHtml(
                    healthRes.ok,
                    'Backend Health',
                    `Status: ${healthRes.status}, Response: ${JSON.stringify(healthData)}`
                ));
            } catch (e) {
                tests.push(getStatusHtml(
                    false,
                    'Backend Health',
                    `Error: ${e.message}. Is the backend running on ${API_URL}?`
                ));
            }
            
            // Test 2: Auth endpoint exists
            try {
                const authRes = await fetch(`${API_URL}/api/v1/auth/login/super`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ email: 'test@test.com', password: 'test' })
                });
                
                tests.push(getStatusHtml(
                    authRes.status !== 404,
                    'Auth Endpoint',
                    `Status: ${authRes.status} (${authRes.statusText})`
                ));
            } catch (e) {
                tests.push(getStatusHtml(
                    false,
                    'Auth Endpoint',
                    `Error: ${e.message}`
                ));
            }
            
            statusDiv.innerHTML = tests.join('');
        }
        
        function decodeToken() {
            const token = localStorage.getItem('sentinelfi_auth_token');
            const resultsDiv = document.getElementById('testResults');
            
            if (!token) {
                resultsDiv.innerHTML = getStatusHtml(false, 'No token found', 'Login first');
                return;
            }
            
            try {
                const parts = token.split('.');
                const header = JSON.parse(atob(parts[0]));
                const payload = JSON.parse(atob(parts[1]));
                
                resultsDiv.innerHTML = `
                    <div class="code-block">
<strong>JWT Header:</strong>
${JSON.stringify(header, null, 2)}

<strong>JWT Payload:</strong>
${JSON.stringify(payload, null, 2)}

<strong>JWT Signature:</strong>
${parts[2]}
                    </div>
                `;
            } catch (e) {
                resultsDiv.innerHTML = getStatusHtml(false, 'Decode Failed', e.message);
            }
        }
        
        async function testSuperAdminEndpoint() {
            const resultsDiv = document.getElementById('testResults');
            const token = localStorage.getItem('sentinelfi_auth_token');
            
            if (!token) {
                resultsDiv.innerHTML = getStatusHtml(false, 'No token', 'Login first');
                return;
            }
            
            resultsDiv.innerHTML = '<p>Testing SuperAdmin endpoint...</p>';
            
            try {
                const res = await fetch(`${API_URL}/api/v1/super/tenants`, {
                    headers: { 'Authorization': `Bearer ${token}` }
                });
                
                const data = await res.json();
                const status = res.ok ? 'pass' : 'fail';
                
                resultsDiv.innerHTML = `
                    <div class="status status-${status}">
                        <div class="status-text">
                            SuperAdmin Endpoint Test
                            <div class="detail">Status: ${res.status} ${res.statusText}</div>
                        </div>
                    </div>
                    <div class="code-block">${JSON.stringify(data, null, 2)}</div>
                `;
            } catch (e) {
                resultsDiv.innerHTML = getStatusHtml(false, 'Request Failed', e.message);
            }
        }

        }
        
        function clearStorage() {
            if (confirm('This will clear all authentication data. Continue?')) {
                localStorage.removeItem('sentinelfi_auth_token');
                localStorage.removeItem('sentinelfi_user');
                alert('Storage cleared! Refresh the page.');
                runDiagnostics();
            }
