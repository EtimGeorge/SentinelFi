export class DashboardController {
  
  /**
   * Get dashboard summary for tenant
   * 
   * This endpoint was returning 500 errors when SuperAdmin users tried to access it.
   * Root cause: The middleware wasn't properly blocking SuperAdmin access.
   * 
   * Now fixed with proper middleware chain that blocks non-tenant users.
   */
  async getSummary(req: Request, res: Response): Promise<void> {
    try {
      console.log('[Dashboard Controller] getSummary called by:', {
        userId: req.user?.id,
        email: req.user?.email,
        role: req.user?.role,
        tenantId: req.user?.tenantId
      });

      // Additional validation (middleware should have already caught this)
      if (!req.user) {
        console.error('[Dashboard Controller] No user in request');
        res.status(401).json({ 
          success: false,
          message: 'Unauthorized' 
        });
        return;
      }

      // SuperAdmin should NEVER reach this point due to middleware
      // But defensive programming is good
      if (req.user.role === Role.SuperAdmin) {
        console.error('[Dashboard Controller] SuperAdmin incorrectly routed to tenant endpoint');
        res.status(403).json({ 
          success: false,
          message: 'SuperAdmin users should use /api/v1/super/dashboard endpoints',
          correctEndpoint: '/api/v1/super/dashboard/global-summary'
        });
        return;
      }

      if (!req.user.tenantId) {
        console.error('[Dashboard Controller] Tenant user missing tenantId');
        res.status(400).json({ 
          success: false,
          message: 'User is not associated with a tenant' 
        });
        return;
      }

      // Fetch tenant-specific data
      const tenantId = req.user.tenantId;
      
      // TODO: Replace with actual database queries
      const summary = await this.fetchDashboardSummary(tenantId);

      console.log('[Dashboard Controller] Summary fetched successfully for tenant:', tenantId);

      res.json({ 
        success: true,
        data: summary 
      });

    } catch (error: any) {
      console.error('[Dashboard Controller] Error in getSummary:', error);
      res.status(500).json({ 
        success: false,
        message: 'Failed to fetch dashboard summary',
        error: process.env.NODE_ENV === 'development' ? error.message : undefined
      });
    }
  }

  /**
   * Get analytics data for tenant

    try {
      console.log('[Dashboard Controller] getAnalytics called by:', req.user?.email);

        res.status(400).json({ 
          success: false,
          message: 'User is not associated with a tenant' 
        });
        return;
      }

      const { period = '30d', metrics = 'all' } = req.query;
      const analytics = await this.fetchAnalytics(req.user.tenantId, period as string, metrics as string);

      res.json({ 
        success: true,
        data: analytics

    } catch (error: any) {
      console.error('[Dashboard Controller] Error in getAnalytics:', error);
      res.status(500).json({ 
        success: false,
        message: 'Failed to fetch analytics',
        error: process.env.NODE_ENV === 'development' ? error.message : undefined

  }

          success: false,

        req.user.tenantId, 
        Number(limit), 
        Number(offset)
      );

      res.json({
